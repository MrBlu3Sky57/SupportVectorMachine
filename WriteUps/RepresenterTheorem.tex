\section* {Representer Theorem}
\textbf{Lemma 1}: A Hilbert space $\mathcal{H}$ of functions from a set $X$ to $\mathbb{R}$ (or $\mathbb{C}$) is a reproducing kernel Hilbert space 
if and only if $\exists K_x \in \mathcal{H}$ such that $\forall f \in \mathcal{H}$ $\langle f, K_x \rangle = f(x)$.
\begin{proof}
$\implies$ By definition a Hilbert space $H$ is a reproducing kernel Hilbert space if $\forall x \in X$ the linear functional $L_x: \mathcal{H} \to \mathbb{R}$
given by $L_x(f) = f(x)$ is continuous. Thus $L_x$ is bounded on $\mathcal{H}$ as continuity implies boundedness, so we can apply Riesz's representation theorem to find a unique $g_x \in \mathcal{H}$
such that $L_x(f) = \langle f, g_x \rangle$. Setting $K_x = g_x$, it is clear that $K_x$ is well defined as $g_x$ is unique by Riesz's and 
$\langle f, K_x \rangle = \langle f, g_x \rangle = f(x)$. Thus, $K_x$ is a reproducing kernel of $\mathcal{H}$.
\\ \\
$\impliedby$ Suppose there exists a $K_x \in \mathcal{H}$ such that $\langle f, K_x \rangle = f(x)$. Then we can define a linear functional $L_x: X \to \mathbb{R}$
$L_x(f) = \langle f, K_x \rangle$. By Cauchy-Schwarz, $|\langle f, K_x \rangle| \leq ||f||_\mathcal{H}^2||K_x||_\mathcal{H}^2$. Thus, $L_x(f)$ is bounded. Now we must
show that $L_x$ is continuous. Given $\epsilon > 0$ pick $\delta = \frac{\epsilon}{||K_x||^2_\mathcal{H}}$. Then for $g \in \mathcal{H}$ such that $||f - g||_\mathcal{H}^2 < \delta$
we have:
\begin{align*}
    ||L_x(f) - L_x(g)|| &= |L_x(f - g)| \text{ as $L_x$ is a linear functional}\\
    &= |\langle f - g, K_x \rangle| \\
    &\text{By Cauchy Scharwz we have:} \\
    &\leq ||f - g||_\mathcal{H}^2 ||K_x||_\mathcal{H}^2 \\
    &< \frac{\epsilon}{||K_x||^2_\mathcal{H}}||K_x||_\mathcal{H}^2 \\
    &= \epsilon
\end{align*}
Thus, $L_x$ is continuous and H is a reproducing kernel Hilbert space as required.
\end{proof}

\textbf{Lemma 2}: We can characterize $K$ in the above lemma as a function $K: X \times X \to \mathbb{R}$ by $K(x, y) = \langle K_x, K_y \rangle$.
\begin{proof}
    By Lemma 1, there exists $K_y \in \mathcal{H}$ such that $\forall f \in \mathcal{H}$ $\langle f, K_y \rangle = f(y)$. If we consider $K_x \in \mathcal{H}$, 
    we can apply this characterization to it. We find that $K_x(y) = \langle K_x, K_y \rangle$. Thus we can define $K(x, y) = \langle K_x, K_y \rangle$ as a function
    from $X \times X$ to $\mathbb{R}$. In particular, this implies that $K$ is symmetric.
\end{proof}

\textbf{Theorem}: Let $\mathcal{H}$ be a reproducing kernel Hilbert space of functions over $\mathbb{R}$ with domain $V$, a finite dimensional vector space over $\mathbb{R}$
and a set of inputs $X = \{x^{(1)}, \ldots, x^{(m)}\} \subset V$ and outputs $Y = \{y^{(1)}, \ldots, y^{(m)}\} \subset \mathbb{R}$. Now define $J: \mathcal{H} \to \mathbb{R}$
by $J(f) = \displaystyle \sum_{i = 1}^m  L(f(x^{(i)}), y^{(i)}) + \lambda||f||^2_\mathcal{H}$ be a regularized risk function with respect to f. Then
any function $f^*$ that minimzes J has the form $f^* = \sum_{i = 1}^m \alpha_iK(x, x^{(i)})$.
\begin{proof}
Given a reproducing kernel Hilbert space $H$ with kernel $K$, it suffices to show that $\forall f \in \mathcal{H}$ we can find $f^* = \sum_{i = 1}^m \alpha_iK(x, x^{(i)})$
such that $J(f) \geq J(f^*)$. Consider the subspace $W = \text{Span}\{K(\cdot, x^{(i)})\}^m_{i=1}$. W is
finite dimensional by construction, so we may write $\mathcal{H} = W \oplus W^\perp$. Thus given $f \in \mathcal{H}$, we can find $f_1 \in W$ and
$f_2 \in W^\perp$ such that $f = f_1 + f_2$. Applying the reproducing property of K to $f_2$ on $x^{(i)}$ we see that:
\begin{align*}
    f_2(x^{(i)}) &= \langle f_2, K_{x^{(i)}} \rangle \\
    &\text{By Lemma 2:} \\
    &= \langle f_2, K(x^{(i)}, \cdot) \rangle \\
    &= \langle f_2, K(\cdot, x^{(i)}) \rangle \\
    &\text{Since $f_2 \in W^\perp$ :} \\
    &= 0
\end{align*}
Thus, $f(x_i) = f_1(x_i)$. Additionally by the pythagoras theorem: $||f||_\mathcal{H}^2 = ||f_1||_\mathcal{H}^2 + ||f_2||_\mathcal{H}^2 \geq ||f_1||_\mathcal{H}^2$
Now setting $f^* = f_1$. It is clear that:
\begin{align*}
    J(f) &= \displaystyle \sum_{i = 1}^m  L(f(x^{(i)}), y^{(i)}) + \lambda||f||^2_\mathcal{H} \\
    &\geq \displaystyle \sum_{i = 1}^m  L(f^*(x^{(i)}), y^{(i)}) + \lambda||f^*||^2_\mathcal{H} \\
    &= J(f^*)
\end{align*}
\end{proof}
\textbf{Note:} We assume L is convex in its first argument and lower semi-continuous to ensure it attains a minimizer in $\mathcal{H}$
